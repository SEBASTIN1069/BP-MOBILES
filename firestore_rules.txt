Firestore security rules
=========================
Firebase console > Firestore Database > "Rules" tab > paste this over
the existing rules > click "Publish".

This lets EVERYONE read products/accessories (so customers can browse
without logging in), but only someone logged in through admin.html
(your Firebase account) can add, edit, or delete anything.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /accessories/{accessoryId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Customer login accounts (added to fix accounts not persisting).
    // These are NOT Firebase Authentication accounts — they're plain
    // documents checked by script.js itself — so there's no request.auth
    // to lock this down with. It stays open, same trust level the old
    // browser-only version had, just shared centrally now instead of
    // living in one browser's storage.
    match /customers/{mobile} {
      allow read, write: if true;
    }
  }
}
