Firestore security rules
=========================
Firebase console > Firestore Database > "Rules" tab > paste this over
the existing rules > click "Publish".

This lets EVERYONE read products/accessories (so customers can browse
without logging in), but only someone logged in through admin.html
(your Firebase account) can add, edit, or delete anything.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /accessories/{accessoryId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
